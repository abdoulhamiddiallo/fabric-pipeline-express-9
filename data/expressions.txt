==================== WH_Shinkansen (une seule fois) ====================

[Nouvelle requête SQL dans l'entrepôt]
CREATE TABLE dbo.line_scores (line_code varchar(20), on_time_pct decimal(5,1), summary varchar(200));

==================== PL_Line (le pipeline enfant) ====================

[Paramètres de PL_Line : nom, type, valeur par défaut]
line_code    String    tokaido
line_name    String    Tōkaidō
file_name    String    departures_tokaido_2026-10-20.csv

[Load departures : source, nom de fichier (contenu dynamique)]
@pipeline().parameters.file_name

[Load departures : destination, nom de table (contenu dynamique)]
dep_@{pipeline().parameters.line_code}

[Let endpoint sync : temps d'attente en secondes]
90

[Score line : script, type Requête (contenu dynamique)]
INSERT INTO dbo.line_scores SELECT '@{pipeline().parameters.line_code}', CAST(100.0 * SUM(CASE WHEN CAST(delay_s AS int) < 60 THEN 1 ELSE 0 END) / COUNT(*) AS decimal(5,1)), CONCAT('@{pipeline().parameters.line_name} ', CAST(100.0 * SUM(CASE WHEN CAST(delay_s AS int) < 60 THEN 1 ELSE 0 END) / COUNT(*) AS decimal(5,1)), ' % à l''heure (', COUNT(*), ' départs, max ', MAX(CAST(delay_s AS int)), ' s)') FROM LH_Shinkansen.dbo.dep_@{pipeline().parameters.line_code}; SELECT 1 AS ok;

==================== PL_Network (le pipeline parent) ====================

[Reset scores : script, type Requête]
DELETE FROM dbo.line_scores; SELECT 1 AS ok;

[For each line : éléments]
@activity('Read lines').output.value

[Run line : paramètre line_code]
@item().code

[Run line : paramètre line_name]
@item().name

@item().file

[Read scores : script, type Requête]
SELECT STRING_AGG(summary, ' | ') WITHIN GROUP (ORDER BY on_time_pct DESC) AS report FROM dbo.line_scores;
